	effectivenessChart[,] mata in både typer och få ut effect
	
	CalculateDamage attacker, defender, move:
		damage = 0
		if move.isSpecial:
			damage = move.atkPower * ( attacker.spAtk / defender.spDef)
			damage /= 20
	
		else:
			damage = move.atkPower * ( attacker.atk / defender.def)
			damage /= 20
			
		
		if attacker.type == move.type:
			damage *= 1.5
			
		damage *= effectivenessChart[move.type, defender.type]
		
		return damage
